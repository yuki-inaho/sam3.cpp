cmake_minimum_required(VERSION 3.14)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(efficient_sam2_prefix_cache LANGUAGES C CXX)
    enable_testing()
endif()
add_library(efficient_sam2_prefix_cache STATIC cache.cpp digest.cpp third_party/sha256/sha256.c)
target_compile_features(efficient_sam2_prefix_cache PRIVATE cxx_std_17)
target_include_directories(efficient_sam2_prefix_cache PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}
    PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
set_target_properties(efficient_sam2_prefix_cache PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
add_executable(test_efficient_sam2_prefix_cache test_cache.cpp)
target_compile_features(test_efficient_sam2_prefix_cache PRIVATE cxx_std_17)
target_link_libraries(test_efficient_sam2_prefix_cache PRIVATE efficient_sam2_prefix_cache Threads::Threads)
add_test(NAME efficient_sam2_prefix_cache COMMAND test_efficient_sam2_prefix_cache ${CMAKE_CURRENT_BINARY_DIR}/prefix-cache-test-data)
add_executable(test_efficient_sam2_prefix_digest test_digest.cpp)
target_compile_features(test_efficient_sam2_prefix_digest PRIVATE cxx_std_14)
target_link_libraries(test_efficient_sam2_prefix_digest PRIVATE efficient_sam2_prefix_cache)
add_test(NAME efficient_sam2_prefix_digest COMMAND test_efficient_sam2_prefix_digest ${CMAKE_CURRENT_BINARY_DIR}/prefix-digest-test.bin)
