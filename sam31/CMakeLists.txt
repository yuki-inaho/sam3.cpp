option(SAM31_USE_BLAS "Use a system CBLAS library for native CPU inference" ON)
if(SAM31_USE_BLAS)
    find_package(BLAS QUIET)
    include(CheckIncludeFileCXX)
    check_include_file_cxx(cblas.h SAM31_HAVE_CBLAS_H)
    if(BLAS_FOUND AND SAM31_HAVE_CBLAS_H)
        target_link_libraries(sam31_native PRIVATE ${BLAS_LIBRARIES})
        target_compile_definitions(sam31_native PRIVATE SAM31_USE_BLAS=1)
        message(STATUS "SAM31 native: CBLAS acceleration enabled (${BLAS_LIBRARIES})")
    else()
        message(WARNING "SAM31 native: CBLAS unavailable; full-size inference will be slow. Install libopenblas-dev and reconfigure with -U SAM31_HAVE_CBLAS_H.")
    endif()
endif()
